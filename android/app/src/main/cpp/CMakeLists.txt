cmake_minimum_required(VERSION 3.22)
project(meetiq LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/whisper.cpp whisper-build)
set(SHERPA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/sherpa-onnx")
add_library(sherpa_c SHARED IMPORTED)
set_target_properties(sherpa_c PROPERTIES IMPORTED_LOCATION "${SHERPA_ROOT}/jniLibs/${ANDROID_ABI}/libsherpa-onnx-c-api.so")
add_library(meetiq SHARED whisper_jni.cpp speakers_jni.cpp)
target_include_directories(meetiq PRIVATE "${SHERPA_ROOT}/include")
target_link_libraries(meetiq PRIVATE whisper sherpa_c android log)
target_link_options(meetiq PRIVATE "-Wl,-z,max-page-size=16384")
